Écrire avec for-end

clear al; clc

V = [1, 2, 3, 4, 5]

for i = 1:5
   V(i) = i;
end
V

V =
     1     2     3     4     5

W = [2, 4, 6, 8, 10]

for i = 1:5
   W(i) = 2*i;
end
W

for i = 2:2:10
   WW(i/2) = i;
end
WW

j = 0;
for i = 2:2:10
   j = j+1;
   WWW(j) = i;
end
WWW

W =
     2     4     6     8    10
WW =
     2     4     6     8    10
WWW =
     2     4     6     8    10

X = [-3, 0, 3, 6, 9, 12]

for i = 1:6
   X(i) = -3 + 3*(i-1);
end
X

X =
    -3     0     3     6     9    12

Somme de 1 à 5 sans utiliser de vecteur

Somme = 0; % initialisation du sommateur
for i = 1:5
   Somme = Somme + i;
end
Somme

Somme =
    15

Factorielle de 5

Factorielle = 1; % initialisation du sommateur
for i = 1:5
   Factorielle = Factorielle*i;
end
Factorielle

Factorielle =
   120

Somme de Z

Z = [10, 23, 37, 41, 50];
Somme = 0; % initialisation du sommateur
for i = 1:length(Z) % écrire 5 est réducteur.
   % length s'applique à tout Z, peu importe la
   % longueur du vecteur.
   Somme = Somme + Z(i);
end
Somme

Somme =
   161

Moyenne de X.

Moyenne = Somme/length(Z) % ne pas écrire 5.

Moyenne =
         32.2

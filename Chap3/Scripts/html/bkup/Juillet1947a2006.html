Températures mensuelles de juillet à Kuujjuaq

clear all; clear all; clc;
Temperature = load('Kuujjuaq.dat');
plot(Temperature,'o-b');
hold on; % afin d'ajouter la courbe suivante
plot(Temperature,'-r');
grid on;
xlabel('ans');
ylabel('T (°C)');
title(['Température moyenne de juillet ',...
   'à Kuujjuaq']);

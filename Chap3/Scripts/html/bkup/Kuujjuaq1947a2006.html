Kuujjuaq de juillet 1947 à 2006

clear all; clear all; clc;
Temperature = load('Kuujjuaq.dat');
annees = 1947:2006;
plot(annees,Temperature,'o-b');
hold on; % afin d'ajouter la courbe suivante
plot(annees,Temperature,'-r');
grid on;
xlabel('année');
ylabel('T (°C)');
title('Kuujjuaq de juillet 1947 à 2006');

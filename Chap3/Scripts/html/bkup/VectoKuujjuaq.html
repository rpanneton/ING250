clear all;clc
T = load('Kuujjuaq.dat');
ndata=length(T);

Méthode traditionnelle

somme=0;
for i=ndata-9:ndata
   somme=somme+T(i);
end
moyenne=somme/10;
disp(['Moyenne des 10 ',...
   'dernières années : ',...
   num2str(moyenne),'°C.'])
Tgrand=0;

Moyenne des 10 dernières années : 12.38°C.

for i=1:ndata
   if(T(i)<moyenne)
      Tgrand=Tgrand+1;
   end
end
disp(['Pendant ',num2str(Tgrand),' des ',...
   num2str(ndata),' ans, la moyenne mensuelle',...
   ' de juillet']);
disp(['a été inférieure à la moyenne des 10 ',...
   'dernières années.']);

Pendant 44 des 59 ans, la moyenne mensuelle de juillet
a été inférieure à la moyenne des 10 dernières années.

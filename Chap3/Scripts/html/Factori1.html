%    Déplacer le « % » des 2 lignes précédentes
%    pour une exécution dans la fenêtre de Matlab.
F=1;
for i=N:-1:1
   F=F*i;
   fprintf('%4.0f',i,F);
   fprintf('\n');
end
fprintf('Factorielle de %.0f = %.0f\n',N,F);

Affichage de la matrice

clear; clc; close all;
M=load('Refract.dat');
fprintf(' Mvo    Sy    Kc     E   Tmax\n');
fprintf('---- ----- ----- ----- -----\n');
for i=1:length(M)
   fprintf('%4.1f',M(i,1));
   fprintf(' %5.0f',M(i,2:5)); % une rangée
   fprintf('\n');
end
fprintf('---- ----- ----- ----- -----\n');

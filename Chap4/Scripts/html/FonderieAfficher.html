Affichage avec 2 boucles

clear all; clc; close all
M = load('Fonderie.txt');
[nRangees,nColonnes] = size(M);

for i = 1:nRangees
   for j = 1:nColonnes
      fprintf('%5.2f  ',M(i,j));
   end
   fprintf('\n');
end
fprintf('- - - - - - - - - - - - - - - - -\n');

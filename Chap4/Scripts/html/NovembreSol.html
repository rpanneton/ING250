   for j = 1:nc
      if(T(i,j)<T(imin,jmin));
         imin = i; jmin = j;
      end
   end
end

Températures du mois en 4 classes

Compteur = [0 0 0 0];
for i = 1:nL
   for j = 1:nc
      if(T(i,j)<0)
         Compteur(1)=Compteur(1)+1;
      elseif(T(i,j)<5)
         Compteur(2)=Compteur(2)+1;
      elseif(T(i,j)<10)
         Compteur(3)=Compteur(3)+1;
      else
         Compteur(4)=Compteur(4)+1;
      end
   end
end
batons(Compteur)
